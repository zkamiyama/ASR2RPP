cmake_minimum_required(VERSION 3.16)
project(asr2rpp_whisper_regions LANGUAGES C CXX)
if(NOT ASR2RPP_WHISPER_SOURCE)
  message(FATAL_ERROR "ASR2RPP_WHISPER_SOURCE is required")
endif()
add_subdirectory("${ASR2RPP_WHISPER_SOURCE}" upstream)
add_executable(asr2rpp-whisper-regions ../whisper_regions.cpp)
target_compile_features(asr2rpp-whisper-regions PRIVATE cxx_std_17)
target_link_libraries(asr2rpp-whisper-regions PRIVATE whisper)
if(MSVC)
  target_compile_options(asr2rpp-whisper-regions PRIVATE /utf-8)
endif()
